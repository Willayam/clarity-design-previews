import {Input,Output,UrlSource,ALL_FORMATS,CanvasSink,CanvasSource,Mp4OutputFormat,BufferTarget} from '../downloads-20261101/mediabunny.min.mjs';
const {canvas,ctx,segment,refine,save}=api;
window.runE2E=async(rounds=5,duration=60)=>{const rows=[],cam=canvas(),cg=cam.getContext('2d'),person=canvas(),pg=person.getContext('2d'),dest=canvas(1920,1080),g=dest.getContext('2d');
for(let round=0;round<rounds;round++){for(let n=0;n<4;n++){const mode=(round+n)%4;const input=new Input({source:new UrlSource('plate.mp4'),formats:ALL_FORMATS}),track=await input.getPrimaryVideoTrack(),target=new BufferTarget(),output=new Output({format:new Mp4OutputFormat({fastStart:'in-memory'}),target}),source=new CanvasSource(dest,{codec:'avc',bitrate:2500000,keyFrameInterval:2});output.addVideoTrack(source,{frameRate:30});await output.start();const times=Array.from({length:duration*30},(_,i)=>i/30),it=new CanvasSink(track,{poolSize:3}).canvasesAtTimestamps(times)[Symbol.asyncIterator]();let prev=null;const parts={decode:0,mask:0,paint:0,encode:0};const started=performance.now();for(let i=0;i<times.length;i++){let t=performance.now();const {value:f}=await it.next();if(!f)throw Error('missing frame '+i);parts.decode+=performance.now()-t;t=performance.now();cg.drawImage(f.canvas,0,1080,1280,720,0,0,1280,720);let a=segment(cam);if(mode===3&&prev){for(let j=0;j<a.length;j++){const w=Math.abs(a[j]-prev[j])<.2?.65:1;a[j]=w*a[j]+(1-w)*prev[j];}}prev=a;const m=refine(cam,a,Math.min(2,mode));parts.mask+=performance.now()-t;t=performance.now();g.drawImage(f.canvas,0,0,1920,1080,0,0,1920,1080);pg.globalCompositeOperation='copy';pg.drawImage(cam,0,0);pg.globalCompositeOperation='destination-in';pg.drawImage(m,0,0,1280,720);g.drawImage(person,1050,590,800,450);parts.paint+=performance.now()-t;t=performance.now();await source.add(i/30,1/30);parts.encode+=performance.now()-t;}await output.finalize();const sec=(performance.now()-started)/1000;rows.push({round,mode,seconds:sec,frames:times.length,bytes:target.buffer.byteLength,parts});await save('e2e-results.json',JSON.stringify({info,duration,rows},null,2));if(round===0)await save(`e2e-${mode}.mp4`,target.buffer);input.dispose();window.state={phase:'e2e',round,mode,seconds:sec};}}
return rows;};
