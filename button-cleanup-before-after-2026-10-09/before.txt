 * Roles in use: default = primary, outline = secondary, ghost = tertiary,

 * link = inline, dashed = the

 * its whole state (muted, primary when hovered or pressed) and never a fill,
 * gold = a marketing page's trial call to action, ink = the site header's Get

 * `secondary` is defined upstream but unused here — on our near-white card it
 * is indistinguishable from `outline`.
 *
 * The face depends on two things besides the variant:
 *
 *   surface  read from the nearest <Surface> (ui/surface.tsx). Hero buttons wear
 *            depth: the primary gradient and the frosted outline, both with a
 *            halo. A panel is the elevated thing, so its controls stay flat. A
 *            bar replaces every variant's fill with a transparent pill.
 *   paint    `data-button-surface="none"` is the escape hatch out of the two hero
 *            faces, for a marketing CTA that styles itself. It leaves ink, size,
 *            focus and the panel and bar treatments alone. Use it sparingly.
 *
 * The faces read tokens declared on the button itself in index.css
 * (`--btn-gradient`, `--btn-shadow`, …), not on :root: a custom property
 * resolves where it is declared, and `[data-record-action]` re-points
 * `--btn-fill` on the element.
 *
 * `data-active` on an outline button tints it, on a hero or a panel surface: a toolbar menu holding a
 * non-default choice, e.g. the Library type filter.
 *
 * Focus: the painted faces and the bar draw depth in box-shadow, which is where
 * shadcn draws its ring, so they take a 2px outline outside the halo instead.
 *
 * Busy: a Button disabled while its own work runs (aria-busy) keeps full
 * strength, because working is not unavailable. The selector carries both
 * states, so it outranks any single-state dimming rule.
 */

type ButtonProps = ButtonPrimitive.Props &
  Omit<VariantProps<typeof buttonVariants>, "surface" | "paint"> & {
    /** The escape hatch out of the hero faces. Typed so it cannot be misspelled. */
    "data-button-surface"?: "none"
  }

function Button({
  className,
  variant = "default",
  size = "default",
  ...props
}: ButtonProps) {
  const surface = useSurface()
  const paint = props["data-button-surface"] === "none" ? "none" : "face"
  return (
    <ButtonPrimitive
      data-slot="button"
      data-button-variant={variant}
      data-button-size={size}
      data-variant={variant}
      data-size={size}
      className={cn(buttonVariants({ variant, size, surface, paint, className }))}
      {...props}
    />
  )
}

export { Button, buttonVariants }
